1    package com.smartsensor.resistancemeter.fragment
2    
3    import android.os.Bundle
4    import android.view.LayoutInflater
5    import android.view.View
6    import android.view.ViewGroup
7    import android.widget.Toast
8    import androidx.fragment.app.Fragment
9    import com.smartsensor.resistancemeter.MainActivity
10   import com.smartsensor.resistancemeter.databinding.FragmentControlBinding
11   import com.smartsensor.resistancemeter.service.BluetoothService
12   import com.smartsensor.resistancemeter.utils.Constants
13   import com.smartsensor.resistancemeter.utils.DataParser
14   
15   class ControlFragment : Fragment() {
16   
17       private var _binding: FragmentControlBinding? = null
18       private val binding get() = _binding!!
19   
20       private var bluetoothService: BluetoothService? = null
21   
22       override fun onCreateView(
23           inflater: LayoutInflater,
24           container: ViewGroup?,
25           savedInstanceState: Bundle?
26       ): View {
27           _binding = FragmentControlBinding.inflate(inflater, container, false)
28           return binding.root
29       }
30   
31       override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
32           super.onViewCreated(view, savedInstanceState)
33   
34           setupClickListeners()
35           setupObservers()
36       }
37   
38       private fun setupClickListeners() {
39           binding.apply {
40               calibrateButton.setOnClickListener { sendCalibrationCommand() }
41               resetButton.setOnClickListener { sendResetCommand() }
42               sendCustomButton.setOnClickListener { sendCustomCommand() }
43           }
44       }
45   
46       private fun setupObservers() {
47           bluetoothService = (activity as? MainActivity)?.getBluetoothService()
48   
49           bluetoothService?.connectionState?.observe(viewLifecycleOwner) { connected ->
50               binding.apply {
51                   connectionStatus.text = if (connected) "设备已连接" else "设备未连接"
52                   calibrateButton.isEnabled = connected
53                   resetButton.isEnabled = connected
54                   sendCustomButton.isEnabled = connected
55               }
56           }
57       }
58   
59       private fun sendCalibrationCommand() {
60           bluetoothService?.let { service ->
61               val command = DataParser.createControlCommand(Constants.CMD_CALIBRATION)
62               if (service.sendData(command)) {
63                   Toast.makeText(context, "校准命令已发送", Toast.LENGTH_SHORT).show()
64               } else {
65                   Toast.makeText(context, "发送失败", Toast.LENGTH_SHORT).show()
66               }
67           }
68       }
69   
70       private fun sendResetCommand() {
71           bluetoothService?.let { service ->
72               val command = DataParser.createControlCommand(Constants.CMD_RESET)
73               if (service.sendData(command)) {
74                   Toast.makeText(context, "复位命令已发送", Toast.LENGTH_SHORT).show()
75               } else {
76                   Toast.makeText(context, "发送失败", Toast.LENGTH_SHORT).show()
77               }
78           }
79       }
80   
81       private fun sendCustomCommand() {
82           val customText = binding.customCommandEdit.text.toString().trim()
83           if (customText.isEmpty()) {
84               Toast.makeText(context, "请输入命令", Toast.LENGTH_SHORT).show()
85               return
86           }
87   
88           bluetoothService?.let { service ->
89               try {
90                   val data = customText.toByteArray(Charsets.UTF_8)
91                   val command = DataParser.createControlCommand(Constants.CMD_CONFIG_SET, data)
92                   if (service.sendData(command)) {
93                       Toast.makeText(context, "自定义命令已发送", Toast.LENGTH_SHORT).show()
94                       binding.customCommandEdit.text?.clear()
95                   } else {
96                       Toast.makeText(context, "发送失败", Toast.LENGTH_SHORT).show()
97                   }
98               } catch (e: Exception) {
99                   Toast.makeText(context, "命令格式错误", Toast.LENGTH_SHORT).show()
100              }
101          }
102      }
103  
104      override fun onDestroyView() {
105          super.onDestroyView()
106          _binding = null
107      }
108  }
