10   import androidx.fragment.app.Fragment
11   import com.smartsensor.resistancemeter.databinding.FragmentSettingsBinding
12   
13   class SettingsFragment : Fragment() {
14   
15       private var _binding: FragmentSettingsBinding? = null
16       private val binding get() = _binding!!
17   
18       private lateinit var preferences: SharedPreferences
19   
20       override fun onCreateView(
21           inflater: LayoutInflater,
22           container: ViewGroup?,
23           savedInstanceState: Bundle?
24       ): View {
25           _binding = FragmentSettingsBinding.inflate(inflater, container, false)
26           return binding.root
27       }
28   
29       override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
30           super.onViewCreated(view, savedInstanceState)
31   
32           preferences = PreferenceManager.getDefaultSharedPreferences(requireContext())
33   
34           setupClickListeners()
35           loadSettings()
36       }
37   
38       private fun setupClickListeners() {
39           binding.apply {
40               saveButton.setOnClickListener { saveSettings() }
41               exportButton.setOnClickListener { exportData() }
42           }
43       }
44   
45       private fun loadSettings() {
46           binding.apply {
47               autoConnectSwitch.isChecked = preferences.getBoolean("auto_connect", false)
48               samplingRateEdit.setText(preferences.getInt("sampling_rate", 10).toString())
49               chartMaxEntriesEdit.setText(preferences.getInt("chart_max_entries", 100).toString())
50           }
51       }
52   
53       private fun saveSettings() {
54           try {
55               val editor = preferences.edit()
56               binding.apply {
57                   editor.putBoolean("auto_connect", autoConnectSwitch.isChecked)
58                   editor.putInt("sampling_rate", samplingRateEdit.text.toString().toInt())
59                   editor.putInt("chart_max_entries", chartMaxEntriesEdit.text.toString().toInt())
60               }
61               editor.apply()
62   
63               Toast.makeText(context, "设置已保存", Toast.LENGTH_SHORT).show()
64           } catch (e: NumberFormatException) {
65               Toast.makeText(context, "请输入有效的数值", Toast.LENGTH_SHORT).show()
66           }
67       }
68   
69       private fun exportData() {
70           // TODO: 实现数据导出功能
71           Toast.makeText(context, "数据导出功能开发中", Toast.LENGTH_SHORT).show()
72       }
73   
74       override fun onDestroyView() {
75           super.onDestroyView()
76           _binding = null
77       }
78   }
