16   object BluetoothUtils {
17   
18       private const val TAG = "BluetoothUtils"
19   
20       fun isBluetoothSupported(context: Context): Boolean {
21           return context.packageManager.hasSystemFeature(PackageManager.FEATURE_BLUETOOTH_LE)
22       }
23   
24       fun getBluetoothAdapter(context: Context): BluetoothAdapter? {
25           val bluetoothManager = context.getSystemService(Context.BLUETOOTH_SERVICE) as BluetoothManager?
26           return bluetoothManager?.adapter
27       }
28   
29       fun isBluetoothEnabled(context: Context): Boolean {
30           val adapter = getBluetoothAdapter(context)
31           return adapter?.isEnabled == true
32       }
33   
34       // 检查位置服务是否开启（仅Android 11及以下需要）
35       fun isLocationEnabled(context: Context): Boolean {
36           if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
37               // Android 12+ 不需要位置服务（如果使用 neverForLocation）
38               return true
39           }
40           val locationManager = context.getSystemService(Context.LOCATION_SERVICE) as LocationManager
41           return LocationManagerCompat.isLocationEnabled(locationManager)
42       }
43   
44       fun hasRequiredPermissions(context: Context): Boolean {
45           val requiredPermissions = getRequiredPermissions()
46           val result = requiredPermissions.all { permission ->
47               ContextCompat.checkSelfPermission(context, permission) == PackageManager.PERMISSION_GRANTED
48           }
49   
50           Log.d(TAG, "权限检查结果: $result")
51           requiredPermissions.forEach { permission ->
52               val granted = ContextCompat.checkSelfPermission(context, permission) == PackageManager.PERMISSION_GRANTED
53               Log.d(TAG, "权限 $permission: $granted")
54           }
55   
56           return result
57       }
58   
59       fun requestPermissions(activity: Activity) {
60           val requiredPermissions = getRequiredPermissions()
61           val missingPermissions = requiredPermissions.filter { permission ->
62               ContextCompat.checkSelfPermission(activity, permission) != PackageManager.PERMISSION_GRANTED
63           }
64   
65           Log.d(TAG, "缺少的权限: $missingPermissions")
66   
67           if (missingPermissions.isNotEmpty()) {
68               ActivityCompat.requestPermissions(
69                   activity,
70                   missingPermissions.toTypedArray(),
71                   Constants.REQUEST_PERMISSIONS
72               )
73           }
74       }
75   
76       private fun getRequiredPermissions(): List<String> {
77           val permissions = mutableListOf<String>()
78   
79           if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
80               // Android 12+ (API 31+)
81               Log.d(TAG, "使用Android 12+权限模型")
82               permissions.addAll(listOf(
83                   Manifest.permission.BLUETOOTH_SCAN,
84                   Manifest.permission.BLUETOOTH_CONNECT,
85                   // 注意：如果在manifest中使用了neverForLocation，就不需要位置权限
86                   // Manifest.permission.ACCESS_FINE_LOCATION
87               ))
88           } else {
89               // Android 11及以下
90               Log.d(TAG, "使用传统蓝牙权限模型")
91               permissions.addAll(listOf(
92                   Manifest.permission.BLUETOOTH,
93                   Manifest.permission.BLUETOOTH_ADMIN
94               ))
95   
96               // Android 6+ 需要位置权限进行BLE扫描
97               if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
98                   if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
99                       // Android 10+ 需要精确位置
100                      permissions.add(Manifest.permission.ACCESS_FINE_LOCATION)
101                  } else {
102                      // Android 6-9 可以使用粗略位置
103                      permissions.add(Manifest.permission.ACCESS_COARSE_LOCATION)
104                  }
105              }
106          }
107  
108          Log.d(TAG, "所需权限: $permissions")
109          return permissions
110      }
111  
112      // 获取权限状态详情（用于调试）
113      fun getPermissionStatus(context: Context): Map<String, Boolean> {
114          val permissions = getRequiredPermissions()
115          return permissions.associateWith { permission ->
116              ContextCompat.checkSelfPermission(context, permission) == PackageManager.PERMISSION_GRANTED
117          }
118      }
119  
120      // 检查是否可以进行BLE扫描
121      fun canStartBleScan(context: Context): Boolean {
122          val hasPermissions = hasRequiredPermissions(context)
123          val bluetoothEnabled = isBluetoothEnabled(context)
124          val locationEnabled = isLocationEnabled(context)
125  
126          Log.d(TAG, "BLE扫描检查 - 权限:$hasPermissions, 蓝牙:$bluetoothEnabled, 位置:$locationEnabled")
127  
128          return hasPermissions && bluetoothEnabled && locationEnabled
129      }
