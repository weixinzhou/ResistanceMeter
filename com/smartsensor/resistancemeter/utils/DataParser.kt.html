1    package com.smartsensor.resistancemeter.utils

3    import android.util.Log
4    import com.smartsensor.resistancemeter.model.ResistanceData
5    import com.smartsensor.resistancemeter.model.DataPacket

7    object DataParser {
8    
9        private const val TAG = "DataParser"
10   
11       // ESP32实际发送的数据包长度
12       private const val ESP32_PACKET_SIZE = 12
13   
14       fun parseResistanceData(data: ByteArray?): ResistanceData? {
15           if (data == null) {
16               Log.w(TAG, "接收到空数据")
17               return null
18           }
19   
20           Log.d(TAG, "========== ESP32数据解析 ==========")
21           Log.d(TAG, "数据长度: ${data.size} (预期: $ESP32_PACKET_SIZE)")
22   
23           // 打印原始数据
24           val hexString = data.joinToString(" ") { "%02x".format(it.toInt() and 0xFF) }
25           Log.d(TAG, "十六进制: $hexString")
26   
27           // 检查数据长度
28           if (data.size != ESP32_PACKET_SIZE) {
29               Log.w(TAG, "数据长度不匹配: ${data.size} != $ESP32_PACKET_SIZE")
30               return null
31           }
32   
33           try {
34               // 1. 验证包头
35               val header = data[0].toInt() and 0xFF
36               if (header != 0xAA) {
37                   Log.w(TAG, "包头错误: 0x%02x != 0xAA".format(header))
38                   return null
39               }
40               Log.d(TAG, "✓ 包头正确: 0xAA")
41   
42               // 2. 验证命令类型
43               val cmd = data[1].toInt() and 0xFF
44               if (cmd != 0x01) {
45                   Log.w(TAG, "命令类型错误: 0x%02x != 0x01".format(cmd))
46                   return null
47               }
48               Log.d(TAG, "✓ 命令类型正确: 0x01")
49   
50               // 3. 解析三个电阻值 (uint16, 小端序, 需要除以100)
51               val r1Raw = ((data[3].toInt() and 0xFF) shl 8) or (data[2].toInt() and 0xFF)
52               val r2Raw = ((data[5].toInt() and 0xFF) shl 8) or (data[4].toInt() and 0xFF)
53               val r3Raw = ((data[7].toInt() and 0xFF) shl 8) or (data[6].toInt() and 0xFF)
54   
55               val resistance1 = r1Raw / 100.0f
56               val resistance2 = r2Raw / 100.0f
57               val resistance3 = r3Raw / 100.0f
58   
59               Log.d(TAG, "电阻原始值: R1=$r1Raw, R2=$r2Raw, R3=$r3Raw")
60               Log.d(TAG, "电阻实际值: R1=%.2fΩ, R2=%.2fΩ, R3=%.2fΩ".format(resistance1, resistance2, resistance3))
61   
62               // 4. 解析时间戳 (uint16, 小端序)
63               val timestampRaw = ((data[9].toInt() and 0xFF) shl 8) or (data[8].toInt() and 0xFF)
64               Log.d(TAG, "时间戳原始值: $timestampRaw")
65   
66               // 5. 解析状态
67               val status = data[10].toInt() and 0xFF
68               Log.d(TAG, "状态: $status")
69   
70               // 6. 验证校验和
71               val receivedChecksum = data[11].toInt() and 0xFF
72               var calculatedChecksum = 0
73               for (i in 0 until 11) {
74                   calculatedChecksum += (data[i].toInt() and 0xFF)
75               }
76               calculatedChecksum = calculatedChecksum and 0xFF // 取低8位
77   
78               Log.d(TAG, "校验和: 接收=0x%02x, 计算=0x%02x".format(receivedChecksum, calculatedChecksum))
79   
80               if (receivedChecksum != calculatedChecksum) {
81                   Log.w(TAG, "校验和错误，但继续处理数据")
82                   // 注意：不返回null，因为数据可能仍然有效
83               } else {
84                   Log.d(TAG, "✓ 校验和正确")
85               }
86   
87               // 7. 验证电阻值合理性
88               if (!isValidResistance(resistance1) || !isValidResistance(resistance2) || !isValidResistance(resistance3)) {
89                   Log.w(TAG, "电阻值超出合理范围: R1=$resistance1, R2=$resistance2, R3=$resistance3")
90                   return null
91               }
92   
93               // 8. 使用当前系统时间作为时间戳
94               val currentTimestamp = System.currentTimeMillis()
95   
96               Log.d(TAG, "✓ ESP32数据解析成功")
97               Log.d(TAG, "=======================================")
98   
99               return ResistanceData(resistance1, resistance2, resistance3, currentTimestamp, status)
100  
101          } catch (e: Exception) {
102              Log.e(TAG, "解析异常: ${e.message}")
103              e.printStackTrace()
104              return null
105          }
106      }
107  
108      // 验证电阻值是否合理 (0-10000Ω)
109      private fun isValidResistance(value: Float): Boolean {
110          return value >= 0f && value <= 10000f && !value.isNaN() && !value.isInfinite()
111      }
112  
113      fun createControlCommand(command: Byte, parameters: ByteArray? = null): ByteArray {
114          val packet = DataPacket(Constants.PACKET_HEADER, command, parameters)
115          return packet.toByteArray()
116      }
